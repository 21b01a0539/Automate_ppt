from datasets import load_dataset
squad = load_dataset("squad")
squad
squad["train"]
squad['train'][0]
squad["validation"][:3]
from transformers import AutoTokenizer, AutoModelForQuestionAnswering
tokenizer = AutoTokenizer.from_pretrained("bert-base-cased")
model = AutoModelForQuestionAnswering.from_pretrained("bert-base-cased")
model
squad["validation"][:1]
text = "Which NFL team represented the AFC at Super Bowl 50?"
inputs = tokenizer(text)
context = squad["validation"][0]["context"]
context
inputs = tokenizer(text,context,return_tensors="pt")
inputs
input_ids = inputs["input_ids"]
attention_mask = inputs["attention_mask"]
import torch
with torch.no_grad():
  outputs = model(input_ids,attention_mask=attention_mask)
outputs
start_scores = outputs.start_logits
end_scores = outputs.end_logits
start_index = torch.argmax(start_scores)
end_index = torch.argmax(end_scores) + 1
answer_token = input_ids[0][start_index:end_index]
answer = tokenizer.decode(answer_token)
answer
